3  Is the Cre signal real? Confound-resistant checks

Analyses that the run cannot drive: the replicate false-positive floor, RUVs, the germline EFAD benchmark, within-Cre dose-response, and the 2025-10-only candidate pathways and genes.

3.1 Confound-resistant checks

Four standard checks on whether a Cre-vs-GFP signal could be technical:

  1. Replicate false-positive floor. The bridge mice (same animal, both runs) are technical replicates, so every “DEG” between their two copies is a batch false positive. Any Cre effect no larger than this floor is indistinguishable from batch.
  2. RUVs (Risso et al. 2014, Nat Biotechnol): unwanted-variation factors estimated from within-group (replicate) variation over empirical control genes, added to the model.
  3. EFAD E2/E2 vs E4/E4 benchmark. The 12-month EFAD mice are all in the 2025-10 run, so their E2-vs-E4 contrast is free of the run confound. Whole-ranked-list agreement with the switch is tested by correlating Wald statistics and by GSEA of the EFAD E2-up / E2-down gene sets on the switch ranking (rather than overlapping thresholded DEG lists).
  4. Within-Cre dose-response. Among Cre mice only, expression regressed on each mouse’s liver APOE E2 allelic fraction (what Cre leaves behind). Batch cannot drive a within-group gradient. The E2 range is narrow (about 0.87-0.96), so each Cre mouse is also left out in turn and only genes that survive every refit are counted.

## 1. replicate false-positive floor -----------------------------------------------------------
replicate_floor <- function(tissue) {
  genes <- gene_qc[[tissue]]$keep
  gm    <- gfp_meta(tissue, dedup = FALSE)      # replicate pairs needed here
  pairs <- gm |> rownames_to_column("sample") |> group_by(mouse_id) |>
    dplyr::filter(n_distinct(run) == 2) |> ungroup()
  if (nrow(pairs) == 0) return(tibble(tissue = tissue, n_pairs = 0L))
  # dispersions from all GFP of the tissue, so even one pair gives a valid test
  dg <- DESeqDataSetFromMatrix(geneCounts[genes, rownames(gm)], gm, ~ sex + run) |>
    norm_dds() |> estimateDispersions(quiet = TRUE)
  pm <- gm[pairs$sample, , drop = FALSE] |> mutate(mouse_id = factor(mouse_id))
  dp <- DESeqDataSetFromMatrix(geneCounts[genes, rownames(pm)], pm,
                               if (nlevels(pm$mouse_id) > 1) ~ mouse_id + run else ~ run)
  if (!is.null(normalizationFactors(dg))) {
    normalizationFactors(dp) <- normalizationFactors(dg)[, colnames(dp), drop = FALSE]
  } else sizeFactors(dp) <- sizeFactors(dg)[colnames(dp)]
  dispersions(dp) <- dispersions(dg)
  dp <- nbinomWaldTest(dp, quiet = TRUE)
  r  <- results(dp, name = run_coef(dp), alpha = PADJ_CUT)
  tibble(tissue = tissue, n_pairs = nlevels(pm$mouse_id),
         fp_genes = list(rownames(r)[!is.na(r$padj) & r$padj < PADJ_CUT]),
         replicate_FP_DEGs = sum(r$padj < PADJ_CUT, na.rm = TRUE),
         replicate_FP_DEGs_abs_lfc_gt1 = sum(r$padj < PADJ_CUT & abs(r$log2FoldChange) > 1, na.rm = TRUE))
}
floor_tbl <- map(TISSUES, replicate_floor) |> bind_rows() |>
  left_join(dplyr::select(de_summary, tissue, Cre_vs_GFP_DEGs = n_sig), by = "tissue")

## 2. RUVs --------------------------------------------------------------------------------------
ruv_check <- function(tissue, k) {
  dds <- fits$fit[[match(tissue, fits$tissue)]]$dds
  m   <- as.data.frame(colData(dds))
  # empirical negative controls: genes least associated with treatment in the primary fit
  res <- fits$fit[[match(tissue, fits$tissue)]]$res
  ctl <- rownames(res)[!is.na(res$pvalue) & res$pvalue > 0.5]
  groups <- RUVSeq::makeGroups(m$treatment)
  ruv <- RUVSeq::RUVs(counts(dds), cIdx = ctl, k = k, scIdx = groups, isLog = FALSE)
  W   <- as.data.frame(ruv$W)
  mw  <- cbind(m, W)
  design_w <- reformulate(c(colnames(W), attr(terms(design(dds)), "term.labels")))
  r <- DESeqDataSetFromMatrix(counts(dds), mw, design_w) |> norm_dds() |> DESeq(quiet = TRUE) |>
    results(name = COEF, alpha = PADJ_CUT)
  tibble(tissue = tissue, k = k, DEGs_RUVs = sum(r$padj < PADJ_CUT, na.rm = TRUE),
         r_stat_vs_primary = cor(r$stat, res$stat, use = "complete.obs"),
         W1_r_with_run = cor(W[[1]], as.numeric(factor(m$run))))
}
ruv_tbl <- expand_grid(tissue = TISSUES, k = 1:2) |> pmap(ruv_check) |> bind_rows()
write_csv(ruv_tbl, file.path(CSV_DIR, "deseq2", "RUVs_sensitivity.csv"))

efad_contrast <- function(tissue) {
  em <- meta[meta$tissue == tissue & meta$arm == "APOE_EFAD" & meta$apoe %in% c("E2", "E4"), ,

  DESeqDataSetFromMatrix(geneCounts[gene_qc[[tissue]]$keep, rownames(em)], em, des) |>

}

benchmark <- imap(efad, \(e, t) {
  sw   <- fits$fit[[match(t, fits$tissue)]]$res
  both <- intersect(rownames(e)[!is.na(e$stat)], rownames(sw)[!is.na(sw$stat)])
  # E2 signature: top 200 genes by EFAD Wald statistic in each direction
  st   <- sort(set_names(e$stat, rownames(e))[both], decreasing = TRUE)
  sets <- bind_rows(tibble(term = "EFAD_E2_up",   gene = head(names(st), 200)),
                    tibble(term = "EFAD_E2_down", gene = tail(names(st), 200)))

         NES_E2_down = g$NES[g$ID == "EFAD_E2_down"], padj_E2_down = g$p.adjust[g$ID == "EFAD_E2_down"])
}) |> bind_rows()

## 4. within-Cre dose-response to the liver E2 allelic fraction -------------------------------
e2_by_mouse <- apoe_calls |>
  left_join(rownames_to_column(meta, "sample") |> dplyr::select(sample, mouse_id, tissue, treatment),
            by = "sample") |>
  dplyr::filter(tissue == "Liver", treatment == "Cre") |>
  dplyr::select(mouse_id, e2_frac = rs7412_T_frac)

dose_response <- function(tissue) {
  cm <- aav_meta(tissue) |> rownames_to_column("sample") |>
    dplyr::filter(treatment == "Cre", run == CRE_RUN) |>
    left_join(e2_by_mouse, by = "mouse_id") |> dplyr::filter(!is.na(e2_frac)) |>
    column_to_rownames("sample") |> mutate(e2_frac_c = (e2_frac - mean(e2_frac)) / sd(e2_frac))
  fit_dose <- function(m) {
    m$e2_frac_c <- as.numeric(scale(m$e2_frac))
    DESeqDataSetFromMatrix(geneCounts[gene_qc[[tissue]]$keep, rownames(m)], m, ~ sex + e2_frac_c) |>
      norm_dds() |> DESeq(quiet = TRUE) |> results(name = "e2_frac_c", alpha = PADJ_CUT)
  }
  r  <- fit_dose(cm)
  # leverage: the E2 range is narrow, so one mouse can create a "gradient". Refit leaving out each
  # Cre mouse; a dose gene counts only if it stays significant, same sign, in every refit.
  sig  <- !is.na(r$padj) & r$padj < PADJ_CUT
  loo  <- map(rownames(cm), \(x) fit_dose(cm[rownames(cm) != x, , drop = FALSE]))
  keep <- purrr::reduce(map(loo, \(l) !is.na(l$padj) & l$padj < PADJ_CUT &
                                  sign(l$log2FoldChange) == sign(r$log2FoldChange)), `&`, .init = sig)
  sw <- fits$fit[[match(tissue, fits$tissue)]]$res
  both <- intersect(rownames(r)[!is.na(r$stat)], rownames(sw)[!is.na(sw$stat)])
  tibble(tissue = tissue, n_cre = nrow(cm),
         e2_range = sprintf("%.2f-%.2f", min(cm$e2_frac), max(cm$e2_frac)),
         dose_DEGs = sum(sig),
         dose_DEGs_robust_to_any_one_mouse = sum(keep),
         most_influential_mouse = rownames(cm)[which.min(map_int(loo, \(l) sum(l$padj < PADJ_CUT, na.rm = TRUE)))],
         spearman_with_Cre_vs_GFP = cor(r[both, "stat"], sw[both, "stat"], method = "spearman"))
}
dose_tbl <- map(TISSUES, dose_response) |> bind_rows()
write_csv(dose_tbl, file.path(CSV_DIR, "deseq2", "within_Cre_dose_response.csv"))

3.2 Switch vs germline APOE (EFAD E2 / E3 / E4)

The EFAD mice carry human APOE E2, E3 or E4 for life (12 months, 5xFAD het, male). They were sequenced in the same 2025-10 run as the switch mice, so comparisons among these samples have no run confound. Three questions:

  1. Germline contrasts (E2 vs E3, E3 vs E4, E2 vs E4): which of them does the switch (Cre vs GFP) resemble? If the reversed liver signature tracks APOE genotype, the switch should relate the same way to E2 vs E3 and E3 vs E4; if it is specific to one line (genetic background), only contrasts involving that line will show it.
  2. Direct comparison: switch Cre and switch GFP vs germline E2/E2 and E4/E4 in one model (~ sex [+ extraction batch] + group). Switch GFP (still E4 in liver) vs germline E4 is the reference for how much the two models differ before any switch (AAV, age at start, sex mix, colony).
  3. E2-likeness score: each sample’s position on the germline E2-vs-E4 axis (mean z of the top-200 E2-up genes minus the top-200 E2-down genes, variance-stabilised data). The EFAD E2 and E4 mice define the axis, so their own scores are optimistic; the switch mice are scored out-of-sample.

cur_meta <- function(tissue)
  meta[meta$tissue == tissue & meta$run == CRE_RUN, , drop = FALSE] |>
  rownames_to_column("sample") |>
  mutate(group = case_when(arm == "AAV" ~ paste("Switch", treatment),
                           TRUE ~ paste("EFAD", apoe))) |>
  column_to_rownames("sample") |>
  mutate(group = factor(group, levels = c("Switch GFP", "Switch Cre", "EFAD E2", "EFAD E3", "EFAD E4")))

# include extraction batch only if every batch level holds more than one group
batch_ok <- function(m) n_distinct(m[[BATCH]]) > 1 &&
  all(tapply(m$group, droplevels(m[[BATCH]]), \(x) n_distinct(x) > 1))

## 1. germline contrasts vs the switch ---------------------------------------------------------
germline_contrast <- function(tissue, a, b) {
  em <- cur_meta(tissue) |> dplyr::filter(group %in% paste("EFAD", c(a, b))) |>
    mutate(group = factor(group, levels = paste("EFAD", c(b, a)))) |> droplevels()
  des <- if (batch_ok(em)) ~ extraction_batch + group else ~ group
  DESeqDataSetFromMatrix(geneCounts[gene_qc[[tissue]]$keep, rownames(em)], em, des) |>
    norm_dds() |> DESeq(quiet = TRUE) |> results(contrast = c("group", paste("EFAD", a), paste("EFAD", b)),
                                   alpha = PADJ_CUT)
}
relate <- function(e, sw, label, tissue) {
  both <- intersect(rownames(e)[!is.na(e$stat)], rownames(sw)[!is.na(sw$stat)])
  st   <- sort(set_names(e$stat, rownames(e))[both], decreasing = TRUE)
  sets <- bind_rows(tibble(term = "up", gene = head(names(st), 200)),
                    tibble(term = "down", gene = tail(names(st), 200)))
  g <- clusterProfiler::GSEA(sort(set_names(sw$stat, rownames(sw))[both], decreasing = TRUE),
                             TERM2GENE = sets, pvalueCutoff = 1, minGSSize = 10, eps = 0,
                             seed = TRUE, verbose = FALSE) |> as.data.frame()
  tibble(tissue = tissue, germline = label,
         germline_DEGs = sum(e$padj < PADJ_CUT, na.rm = TRUE),
         spearman_with_switch = cor(e[both, "stat"], sw[both, "stat"], method = "spearman"),
         NES_up = g$NES[g$ID == "up"], padj_up = g$p.adjust[g$ID == "up"],
         NES_down = g$NES[g$ID == "down"], padj_down = g$p.adjust[g$ID == "down"])
}
germ_pairs <- tribble(~a, ~b, "E2", "E4", "E2", "E3", "E3", "E4")
germline <- expand_grid(tissue = TISSUES, germ_pairs) |>

## 2. direct comparison: switch groups vs germline E2 / E4 ------------------------------------
direct <- map(set_names(TISSUES), \(t) {
  dm  <- cur_meta(t) |> dplyr::filter(group != "EFAD E3") |> droplevels()
  des <- if (batch_ok(dm)) ~ sex + extraction_batch + group else ~ sex + group
  dds <- DESeqDataSetFromMatrix(geneCounts[gene_qc[[t]]$keep, rownames(dm)], dm, des) |>
    norm_dds() |> DESeq(quiet = TRUE)
  cmp <- list(c("Switch Cre", "EFAD E2"), c("Switch Cre", "EFAD E4"),
              c("Switch GFP", "EFAD E4"), c("Switch GFP", "EFAD E2"))
  map(cmp, \(k) {
    r <- results(dds, contrast = c("group", k), alpha = PADJ_CUT)
    tibble(tissue = t, comparison = paste(k, collapse = " vs "),
           design = paste(deparse(des), collapse = ""),
           DEGs = sum(r$padj < PADJ_CUT, na.rm = TRUE),
           DEGs_abs_lfc_gt1 = sum(r$padj < PADJ_CUT & abs(r$log2FoldChange) > 1, na.rm = TRUE))
  }) |> bind_rows()
}) |> bind_rows()

3.3 Candidate pathways (2025-10 run only)

Pathways that come up in the liver GSEA (Cre vs GFP) but rest on few controls: OXPHOS up, ribosome biogenesis up, cell-substrate adhesion down, negative regulation of immune response down. They fail the strict leave-one-out significance test (dropping one 2025-10 GFP mouse lifts padj above 0.05) while keeping their direction, so they are followed up here without the 2025-04 samples at all:

  • per-sample pathway scores (mean z, variance-stabilised counts) in 2025-10-run livers only: 10 Cre, 3 GFP and 9 EFAD (E2 / E3 / E4), one sample per mouse, no run confound;
  • Cre vs the switch GFP mice (same colony and run; Welch t-test), and, as a secondary reference, vs all other 2025-10 livers including EFAD (Wilcoxon, with each GFP left out);
  • technical flag: correlation of each score with STAR uniquely-mapped % within the Cre group;
  • immune term: is it a regulation change or a Kupffer / leukocyte composition change?

The EFAD mice are a different colony (male, 12 months) and were not designed as controls for the switch, so they widen the reference but are not matched to it. These pathways were found in this data, so this is support, not independent replication.

score_tissue <- function(tissue) {

    rownames_to_column("sample") |>
    mutate(group = if_else(arm == "AAV", paste("Switch", treatment), paste("EFAD", apoe)),

                                                 rownames(z)), , drop = FALSE]), .before = 1)
}

3.4 Candidate genes (2025-10 run only)

The candidate-pathway checks applied to every gene, to see whether any single gene could survive the batching and be worth following up. Each gene is tested in the 2025-10 run only, so the run cannot drive it: Cre vs the switch GFP mice, in a model that also holds the 2025-10 EFAD mice (~ sex [+ extraction batch] + group, group = Cre / GFP / EFAD E2 / E3 / E4). The EFAD mice only add to the dispersion estimates; they are never the comparison, because they differ from the switch colony (male, 12 months, different line) and would make colony look like Cre. Each gene is then checked against:

  1. Leave-one-GFP-out: still padj < 0.05, same direction, with each 2025-10 GFP mouse removed.
  2. Main model: same direction in the batch-corrected Cre vs GFP model (all controls).
  3. Not a replicate false positive: not different between the two runs of the same mouse.
  4. Not tracking library quality: |r| < 0.6 with STAR uniquely-mapped % within the Cre mice.

Tier 1 = passes everything; Tier 2 = significant in the 2025-10 test with the right direction in check 2, but fails leave-one-out or a technical flag. Brain is the negative control: the AAV-TBG switch acts in hepatocytes, so the number of brain genes that pass is an estimate of how many liver hits the pipeline lets through by chance or artifact.

candidate_gene_tissue <- function(tissue) {
  genes <- gene_qc[[tissue]]$keep
  om <- meta[meta$tissue == tissue & meta$run == CRE_RUN, , drop = FALSE] |>
    mutate(group = factor(if_else(arm == "AAV", paste0("Switch_", treatment),
                                  paste0("EFAD_", apoe)),
                          levels = c("Switch_GFP", "Switch_Cre", "EFAD_E2", "EFAD_E3", "EFAD_E4")),
           cre = group)
  des_for <- function(m) if (n_distinct(m[[BATCH]]) > 1 &&
                             all(tapply(m$group, droplevels(m[[BATCH]]), \(x) n_distinct(x) > 1)))
    ~ sex + extraction_batch + group else ~ sex + group
  fit_oct <- function(m) DESeqDataSetFromMatrix(geneCounts[genes, rownames(m)], m, des_for(m)) |>
    norm_dds() |> DESeq(quiet = TRUE) |>
    results(contrast = c("group", "Switch_Cre", "Switch_GFP"), alpha = PADJ_CUT)
  r   <- fit_oct(om)
  gfp <- rownames(om)[om$arm == "AAV" & om$treatment == "GFP"]
  loo <- map(gfp, \(x) fit_oct(om[rownames(om) != x, , drop = FALSE] |> droplevels()))
  loo_ok <- purrr::reduce(map(loo, \(l) !is.na(l$padj) & l$padj < PADJ_CUT &
                                         sign(l$log2FoldChange) == sign(r$log2FoldChange)), `&`)
  main   <- fits$fit[[match(tissue, fits$tissue)]]
  # library-quality tracking within the Cre mice (normalised log counts vs STAR unique %)
  cre_s <- rownames(om)[om$group == "Switch_Cre"]
  lc <- log2(counts(norm_dds(DESeqDataSetFromMatrix(geneCounts[genes, cre_s], om[cre_s, ], ~ 1)),
                    normalized = TRUE) + 1)
  uq <- star_unique$star_unique_pct[match(cre_s, star_unique$sample)]
  r_tech <- apply(lc, 1, \(x) suppressWarnings(cor(x, uq)))
  fp <- floor_tbl$fp_genes[[match(tissue, floor_tbl$tissue)]] %||% character()
  tibble(ensemblID = rownames(r), tissue = tissue,
         log2FC_oct = r$log2FoldChange, padj_oct = r$padj,
         loo_ok = loo_ok,
         log2FC_main = main$res[rownames(r), "log2FoldChange"], padj_main = main$res[rownames(r), "padj"],

         replicate_fp = ensemblID %in% fp,
         r_with_lib_quality = r_tech[rownames(r)]) |>
    mutate(sig_oct = !is.na(padj_oct) & padj_oct < PADJ_CUT,
           agree_main = sign(log2FC_main) == sign(log2FC_oct),
           tech_ok = !replicate_fp & abs(r_with_lib_quality) < 0.6,
           tier = case_when(sig_oct & agree_main %in% TRUE & loo_ok & tech_ok ~ "Tier 1",
                            sig_oct & agree_main %in% TRUE ~ "Tier 2",
                            TRUE ~ NA_character_)) |>
    left_join(dplyr::select(id2name, ensemblID, Symbol, biotype), by = "ensemblID") |>
    left_join(cand_member, by = "ensemblID") |>
    relocate(tissue, ensemblID, Symbol, tier)
}
