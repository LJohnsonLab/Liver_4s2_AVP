## ---- analysis settings -----------------------------------------------------------------
TISSUES    <- c("Liver", "Brain")
# Gene counts used for every analysis: "STAR" (STAR --quantMode GeneCounts, unstranded column)
# or "HISAT2" (HISAT2 + htseq-count intersection-strict, the tutorial route). The other one is
# kept as the internal comparison in ALIGNER COMPARISON.
COUNT_SOURCE <- "STAR"
# Normalisation for every DESeq2 fit: "cqn" = conditional quantile normalisation of gene GC
# content and length (Hansen et al. 2012) as DESeq2 normalisation factors; "deseq2" = median-of-
# ratios size factors. cqn was chosen from the BATCH-CORRECTION BENCHMARK: it halved the
# same-mouse difference between runs without using the double-run mice and kept agreement with
# the unconfounded 2025-10-only comparison.
NORMALIZATION <- "cqn"
COUNT_FILES  <- c(STAR = "star_gene_counts.tsv", HISAT2 = "gene_read_counts_table_all_final.tsv")
PADJ_CUT   <- 0.05          # the ONE significance cut-off: every DE, GSEA, ORA and check uses padj < 0.05

## ---- sample QC --------------------------------------------------------------------------
# Absolute cut-offs; a sample is also flagged if it is > QC_MAD MADs worse than its tissue's
# median (below for QC_MIN metrics, above for QC_MAX metrics).
QC_MIN <- c(raw_read_pairs = 20e6, hisat2_overall_alignment_pct = 80,
            hisat2_concordant_unique_pct = 70, pct_assigned_to_genes = 50,
            htseq_assigned = 10e6, detected_genes = 10000, rin = 6)
QC_MAX <- c(pct_mito = 30, pct_rrna = 5, pct_top20 = 40)
QC_MAD <- 3
# Samples removed from every downstream step. Flags never drop anything automatically:
# review the QC plots, then list sample IDs here with the reason.
QC_EXCLUDE <- c(B13 = paste("sample identity: listed as SAD_326 but reads APOE E2/E2; SAD_326's",
                            "2025-04 brain (R1_B1) and both livers read E4/E4"))

## ---- gene QC ----------------------------------------------------------------------------
# "strict": technical biotypes removed + CPM filter in >= smallest-group samples (default)
# "tutorial": rnabio.org filter only (>= 10 reads in >= 1 sample)
GENE_FILTER   <- "strict"
MIN_COUNT     <- 10          # reads; also sets the CPM cut-off (MIN_COUNT / median library size)
MIN_TOTAL     <- 15          # minimum summed count across the tissue's samples
DROP_BIOTYPES <- c("rRNA", "rRNA_pseudogene", "Mt_rRNA", "Mt_tRNA", "misc_RNA",
                   "scaRNA", "snoRNA", "snRNA", "ribozyme", "TEC")
# Predicted / unnamed mouse genes, by symbol: Gm#####, ...Rik, LOC#####, cDNA-clone names
# (BC######, AI######, ...). Genes with no symbol (symbol = Ensembl ID) are dropped too.
DROP_PREDICTED <- "^Gm\\d+$|Rik\\d*$|^LOC\\d+$|^(AA|AB|AC|AF|AI|AU|AV|AW|AY|BC)\\d{5,6}$"
# Cytosolic ribosomal proteins (Rpl#, Rps#, Rplp#, Rpsa and their -ps pseudogenes). Written so
# it does NOT catch Rps6ka*/Rps6kb* kinases or the mitochondrial Mrpl/Mrps genes.
DROP_RIBO      <- "^Rp[ls]\\d+[a-z]?\\d*(-ps\\d+)?$|^Rp[ls]\\d+(al|rt)$|^Rplp\\d|^Rpsa"
## ---- comparison design ------------------------------------------------------------------
# PRIMARY (one sample per mouse; see aav_meta): 2025-10 Cre vs GFP, with the 2025-04 GFP mice
# that were not re-sequenced added as extra controls and the run
# difference FULLY corrected: one DESeq2 GLM over all three sample sets with a batch term
# (~ sex + batch + treatment; batch = extraction batch where every 2025-10 batch holds a GFP
# mouse, else sequencing run -- see design_for()). The batch term removes the run difference and
# carries its uncertainty; the 2025-04 GFP add replication to the dispersion estimates, while the
# Cre-vs-GFP fold change rests on the 2025-10 GFP mice. Rejected alternatives:
#  * apeglm-shrunk run offset from the bridge mice: ~80% of its DEGs followed the raw run
#    difference and the brain showed >2,000 DEGs plus a hAPOE drop (run effects leaking);
#  * a fixed unshrunk offset from all GFP as normalization factors: treats the run correction as
#    known exactly, so its p-values were far too small (2,966 liver DEGs vs 18 within run).
CRE_RUN    <- "2025-10"
GFP_RUN    <- "2025-04"
COEF       <- "treatment_Cre_vs_GFP"
# CHECKS, using the GFP mice sequenced in 2025-10 (3 liver; 2 brain after excluding B13):
#  1. out-of-sample correction check: each 2025-10 GFP mouse in turn is left out of the offset
#     estimate, then compared with the corrected 2025-04 GFP. Few DE genes = the correction
#     generalises to a mouse it did not see. The uncorrected run DE count is shown for scale.
#  2. within-run check: 2025-10 Cre vs 2025-10 GFP only (no 2025-04 samples); used for
#     direction.
#  3. leave-one-out: the primary model refit without each 2025-10 GFP mouse, so no result
#     hinges on one batch-control animal.
# A primary DEG / GSEA term is flagged "robust" if it stays significant in every leave-one-out
# and its direction agrees with the within-run check.
BATCH      <- "extraction_batch"     # 2025-10 extraction batches 1 / 2; R1 = 2025-04
# Candidate pathways (from the liver GSEA; followed up in CANDIDATE PATHWAYS and highlighted on
# the volcano plots). OXPHOS uses nuclear-encoded genes only: the 11 mitochondrially encoded
# genes track read quality (STAR unique %) and are scored separately as a technical flag.
CANDIDATE_TERMS <- c(`OXPHOS (nuclear)`          = "GO:0006119",
                     `Ribosome biogenesis`       = "GO:0042254",
                     `Cell-substrate adhesion`   = "GO:0031589",

    theme(text = element_text(colour = "black"),
          axis.text = element_text(size = base - 1, colour = "black"),
          axis.line = element_line(linewidth = 0.3),
          axis.ticks = element_line(linewidth = 0.3),
          strip.background = element_blank(),
          strip.text = element_text(size = base, face = "bold"),
          legend.text = element_text(size = base - 1),
          legend.key.size = unit(3, "mm"),
          plot.title = element_text(size = base, face = "bold"),
          plot.margin = margin(2, 2, 2, 2, "mm"))
}

save_plot <- function(p, stem, width = "single", height = 70) {
  w <- if (is.character(width)) MM[[width]] else width
  ggsave(paste0(stem, ".pdf"), p, width = w, height = height, units = "mm",
         device = cairo_pdf, family = FONT)
  ggsave(paste0(stem, ".png"), p, width = w, height = height, units = "mm",
         dpi = 300, bg = "white")
  invisible(stem)
}

# pheatmap draws to a device; render it to pdf + png at print size
save_pheatmap <- function(ph, stem, width = "single", height = 85) {
  w <- (if (is.character(width)) MM[[width]] else width) / 25.4
  cairo_pdf(paste0(stem, ".pdf"), width = w, height = height / 25.4, family = FONT)
  grid::grid.newpage(); grid::grid.draw(ph$gtable); dev.off()
  png(paste0(stem, ".png"), width = w, height = height / 25.4, units = "in", res = 300)
  grid::grid.newpage(); grid::grid.draw(ph$gtable); dev.off()
  invisible(stem)
}

# iCre: the AAV is EGFP-T2A-iCre. iCre (codon-improved Cre, GenBank AY056050) is not the
# wild-type Cre contig in the reference, so its reads were counted separately from each sample's
# unaligned reads (12_icre_count on MCC). Reads per million primary reads.
icre_file <- file.path(IN_DIR, "mcc_results", "apoe_transgene_check", "construct_unaligned_reads.tsv")
icre <- if (file.exists(icre_file)) {
  x <- read_tsv(icre_file, show_col_types = FALSE) |>
    pivot_wider(names_from = contig, values_from = reads, values_fill = 0)
  if (!"iCre_AY056050" %in% names(x)) x$iCre_AY056050 <- 0   # no sample had an iCre read
  transmute(x, sample, iCre_reads = iCre_AY056050,
            iCre_rpm = 1e6 * iCre_reads / `__primary_reads`)
} else tibble(sample = character(), iCre_reads = double(), iCre_rpm = double())
write_csv(icre, file.path(CSV_DIR, "qc", "icre_reads.csv"))

# Any sample whose construct / APOE readout contradicts its label. Both AAV arms carry EGFP
# (GFP-only vs EGFP-T2A-iCre), so EGFP marks transduction and iCre / the liver E2 switch mark
# the Cre arm. iCre mRNA from a single AAV-TBG dose declines over months, so low iCre alone is
# not a failure; the APOE E2 allelic fraction (what Cre leaves behind) is the switch readout.
construct_tpm |>
  dplyr::select(sample, tissue, group, gene, tpm) |>
  pivot_wider(names_from = gene, values_from = tpm) |>
  left_join(dplyr::select(apoe_calls, sample, apoe_call = call, e2_frac = rs7412_T_frac),
            by = "sample") |>
  left_join(icre, by = "sample") |>
  mutate(flag = case_when(
    group %in% c("GFP Control", "Cre-AAV") & tissue == "Liver" & EGFP == 0 ~ "AAV label, no EGFP",
    group == "GFP Control" & iCre_rpm > 0.5                                ~ "GFP label, iCre reads",
    group == "Cre-AAV" & tissue == "Liver" & e2_frac < 0.5                 ~ "Cre label, liver not switched",
    group == "GFP Control" & tissue == "Liver" & e2_frac > 0.5             ~ "GFP label, liver switched",
    str_detect(group, "^E\\d") & str_sub(group, 1, 5) != apoe_call         ~ "APOE genotype mismatch",
    TRUE ~ "")) |>
  dplyr::select(sample, tissue, group, EGFP, iCre_reads, iCre_rpm, hAPOE, apoe_call, e2_frac, flag) |>
  arrange(desc(flag != ""), tissue, group)

qc_raw <- read_tsv(file.path(IN_DIR, "mcc_results", "sample_qc_metrics.tsv"), show_col_types = FALSE)

mito_genes <- id2name$ensemblID[id2name$contig == "MT"]
rrna_genes <- id2name$ensemblID[id2name$biotype %in% c("rRNA", "rRNA_pseudogene", "Mt_rRNA")]
lib        <- colSums(geneCounts)
frac_of    <- function(genes) 100 * colSums(geneCounts[intersect(genes, rownames(geneCounts)), , drop = FALSE]) / lib

count_qc <- tibble(
  sample         = colnames(geneCounts),
  detected_genes = colSums(geneCounts >= 5),
  pct_mito       = frac_of(mito_genes),
  pct_rrna       = frac_of(rrna_genes),
  # share of reads taken by the 20 most-expressed genes (library complexity)
  pct_top20      = 100 * map_dbl(colnames(geneCounts), \(s) sum(head(sort(geneCounts[, s], TRUE), 20))) / lib)

qc <- qc_raw |>
  left_join(count_qc, by = "sample") |>
  left_join(rownames_to_column(meta, "sample") |> dplyr::select(sample, tissue, group, treatment, sex, rin),
            by = "sample")

qc_long <- qc |>
  dplyr::select(sample, tissue, group, all_of(c(names(QC_MIN), names(QC_MAX)))) |>
  pivot_longer(-c(sample, tissue, group), names_to = "metric", values_to = "value") |>
  group_by(tissue, metric) |>
  mutate(mad_z    = (value - median(value, na.rm = TRUE)) / mad(value, na.rm = TRUE),
         lo       = unname(QC_MIN[metric]), hi = unname(QC_MAX[metric]),
         fail_abs = (!is.na(lo) & value < lo) | (!is.na(hi) & value > hi),
         # one-sided: only the bad direction counts (deep libraries are not outliers)
         fail_mad = is.finite(mad_z) &
                    ((metric %in% names(QC_MIN) & mad_z < -QC_MAD) |
                     (metric %in% names(QC_MAX) & mad_z >  QC_MAD)),
         flag     = fail_abs | fail_mad) |>
  ungroup()

qc_flags <- qc_long |>
  dplyr::filter(flag) |>
  mutate(reason = sprintf("%s=%s%s", metric, signif(value, 3),
                          if_else(fail_abs, " (cut-off)", sprintf(" (%.1f MAD)", mad_z)))) |>
  group_by(sample, tissue, group) |>
  summarise(n_flags = n(), reasons = paste(reason, collapse = "; "), .groups = "drop") |>
  arrange(desc(n_flags))

write_csv(qc, file.path(CSV_DIR, "qc", "sample_qc_metrics.csv"))
write_csv(qc_flags, file.path(CSV_DIR, "qc", "sample_qc_flags.csv"))

# Cross-tissue contamination / mislabelling: expression of canonical markers of the OTHER tissue.
# Hepatocyte: secreted plasma proteins; brain: neuronal and oligodendrocyte genes. A sample is
# flagged if its other-tissue marker CPM is > 10x its own tissue's median (standard marker-based

1.6 Gene QC

Applied per tissue (AAV mice) before DESeq2:

  1. construct contigs (Cre, EGFP, hAPP, hPSEN1) removed, because they differ by design;
  2. technical / non-polyA biotypes (DROP_BIOTYPES: rRNA, mt-rRNA/tRNA, snoRNA, …) removed;
  3. predicted / unnamed genes (DROP_PREDICTED: Gm, Rik, LOC, cDNA clones, no symbol) and cytosolic ribosomal-protein genes (DROP_RIBO: Rpl, Rps) removed;
  4. expression filter. strict (default): CPM ≥ MIN_COUNT / median library size (in millions) in at least as many samples as the smallest group, and ≥ MIN_TOTAL reads in total (the edgeR::filterByExpr rule). tutorial: ≥ 10 reads in ≥ 1 sample.

filter_genes <- function(tissue) {
  m   <- meta[meta$tissue == tissue & meta$arm == "AAV", , drop = FALSE]
  cts <- geneCounts[, rownames(m), drop = FALSE]
  lib <- colSums(cts)
  min_cpm <- MIN_COUNT / (median(lib) / 1e6)
  n_min   <- min(table(m$treatment))
  cpm     <- t(t(cts) / lib * 1e6)

  genes <- tibble(ensemblID   = rownames(cts),
                  Symbol      = id2name$Symbol[match(rownames(cts), id2name$ensemblID)],
                  biotype     = id2name$biotype[match(rownames(cts), id2name$ensemblID)],
                  total       = rowSums(cts),
                  n_min_reads = rowSums(cts >= MIN_COUNT),
                  n_cpm_ok    = rowSums(cpm >= min_cpm)) |>
    mutate(reason = case_when(
      ensemblID %in% CONSTRUCT_GENES                                    ~ "construct contig",
      biotype %in% DROP_BIOTYPES                                        ~ "technical biotype",
      is.na(Symbol) | Symbol == "" | Symbol == ensemblID |
        str_detect(Symbol, DROP_PREDICTED)                              ~ "predicted gene",
      str_detect(Symbol, DROP_RIBO)                                     ~ "ribosomal protein",
      GENE_FILTER == "tutorial" & n_min_reads < 1                       ~ "low expression",
      GENE_FILTER == "strict" & (n_cpm_ok < n_min | total < MIN_TOTAL)  ~ "low expression",
      TRUE                                                              ~ "kept"))
  list(tissue = tissue, genes = genes, keep = genes$ensemblID[genes$reason == "kept"],
       min_cpm = min_cpm, n_min = n_min)
}

gene_qc <- map(TISSUES, filter_genes) |> set_names(TISSUES)

# log-CPM density of every sample before vs after filtering, with the CPM cut-off
logcpm_long <- function(g, stage) {
  m    <- meta[meta$tissue == g$tissue & meta$arm == "AAV", , drop = FALSE]
  cts  <- geneCounts[, rownames(m), drop = FALSE]
  lib  <- colSums(cts)
  if (stage == "after") cts <- cts[g$keep, , drop = FALSE]
  lcpm <- log2(t(t(cts) / lib * 1e6) + 0.5)
  as_tibble(lcpm, rownames = "gene") |>
    pivot_longer(-gene, names_to = "sample", values_to = "logcpm") |>
    mutate(tissue = g$tissue, stage = stage, cut = log2(g$min_cpm + 0.5))
}
